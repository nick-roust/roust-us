<script>
  const updateSection = () => {
    const current = window.location.hash || '#overview';
    document.querySelectorAll<HTMLAnchorElement>('[data-section-link]').forEach((link) => {
      if (link.hash === current) link.setAttribute('aria-current', 'location');
      else link.removeAttribute('aria-current');
    });
  };
  updateSection();
  window.addEventListener('hashchange', updateSection);
  document.querySelectorAll<HTMLAnchorElement>('.product-navigation--compact a').forEach((link) => {
    link.addEventListener('click', (event) => {
      const target = link.hash && document.getElementById(link.hash.slice(1));
      link.closest('details')?.removeAttribute('open');
      if (target) {
        event.preventDefault();
        history.pushState(null, '', link.hash);
        updateSection();
        requestAnimationFrame(() => {
          target.scrollIntoView();
          target.setAttribute('tabindex', '-1');
          target.focus({ preventScroll: true });
        });
      }
    });
  });
</script>
